Albums

Album (varchar) - Artist (varchar) - Date (date) - Runtime in minutes (smallint) (round up) - Runtime in hours:minutes:seconds (char(9)) (Xh:XXm:XXs)

CREATE TABLE introsql (
name					varchar(100)	NOT NULL ,
artist					varchar(100)	NOT NULL ,
release_date			date			NOT NULL ,
duration_in_minutes		smallint		NOT NULL ,
duration_in_hms			char(8)			NOT NULL ,
CONSTRAINT album_name PRIMARY KEY ( name )
);
